```tsx
interface BilingualFieldsProps {
  fieldName: string;
  labelKeyEl: string;   // i18n key string, e.g. "admin.exercises.titleEl"
  labelKeyEn: string;   // i18n key string, e.g. "admin.exercises.titleEn"
  required?: boolean;
  textarea?: boolean;
  errors?: Record<string, string>;

// Internally it does:
```
