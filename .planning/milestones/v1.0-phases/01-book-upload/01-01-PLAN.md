**2. Add `uploadStreamToCloudinary` to `app/utils/cloudinary.server.ts`:**

Add a new exported function that streams file data directly to Cloudinary using `upload_stream` and Remix's `writeAsyncIterableToWritable`. This avoids loading the entire file into memory (critical for 100MB PDFs).

```typescript
import { writeAsyncIterableToWritable } from "@remix-run/node";

export async function uploadStreamToCloudinary(
  data: AsyncIterable<Uint8Array>,
  options: {
    folder: string;
    resource_type: "raw" | "image";
    tags?: string[];
    public_id?: string;
  }
): Promise<CloudinaryUploadResult> {
  return new Promise((resolve, reject) => {
    const uploadStream = cloudinary.uploader.upload_stream(
      {
        folder: options.folder,
        resource_type: options.resource_type,
        tags: options.tags,
        public_id: options.public_id,
        access_mode: "public",
      },
      (error, result) => {
        if (error) {
          console.error("Cloudinary stream upload error:", error);
          reject(new Error(`Cloudinary upload failed: ${error.message}`));
        } else if (result) {
          resolve({
            public_id: result.public_id,
            secure_url: result.secure_url,
            bytes: result.bytes,
            format: result.format,
            resource_type: result.resource_type,
            created_at: result.created_at,
          });
        }
      }
    );
    writeAsyncIterableToWritable(data, uploadStream).catch(reject);
  });
}
```

Add the `writeAsyncIterableToWritable` import at the top of the file alongside any existing `@remix-run/node` imports. The `cloudinary` instance is already available in the module scope.
